# 任务流管理系统 M1 — 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把"出仓→卸货"全流程用一个独立的作业工单系统管起来,每步留证据,落到具体的司机+车。**M1 范围**:staff/vehicles 档案 + 建单+OCR明细 + 指派 + 主线 6 态推进 + 各阶段拍照 + 签收完成 + 司机退单 + 任务作废 + 整单退回 + task_events 流水时间线。打码闸门(M2)与地图/M4 RBAC 留到后续。

**Architecture:** 完全独立的新模块,不改动现有出货/入库/装柜。复用 `blueprints/_helpers.py` 的 OCR/YPP/mismatch/图片上传逻辑(抽成公共函数);权限判断收敛到一个 `can()` 辅助函数;状态机用 7 个状态 + `task_events` 流水表做证据链;数据入参全靠"拍照/截图 + OCR",与管家婆无侵入衔接(详见 spec §1.1)。

**Tech Stack:** Python 3.12 + Flask 3.1.3 + SQLite 3.45.3 + Tailwind CSS (内联) + 原生 JS。OCR 复用现有 PaddleOCR + DeepSeek(出库页)。不需要任何新依赖。

**Spec:** `docs/superpowers/specs/2026-07-16-任务流管理-design.md`(本计划的唯一需求源)

import base64
import os
import json
from openai import OpenAI

AI_RECOGNIZE_PROMPT = """你是商品单据识别助手。读取图片,严格按 JSON 返回:
{
  "source_order_no": "单据上的订单号(必填)",
  "customer": "客户名(可空)",
  "dest_address": "送货地址(可空)",
  "items": [
    {"product_name":"...","specification":"...","quantity":123,"unit":"y/支/件...","remark":"..."}
  ]
}
注意:remark 中若含 'X支' 或 'X散码' 字样,务必保留;quantity 必须是数字。"""

def recognize_order_image(image_bytes: bytes) -> dict:
    """统一 OCR + DeepSeek 识别入口,被 shipping/任务流 复用。
    返回 dict: {success, source_order_no, customer, dest_address, items, error}"""
    try:
        b64 = base64.b64encode(image_bytes).decode()
        client = OpenAI(
            api_key=os.environ.get("MOONSHOT_API_KEY", ""),
            base_url="https://api.moonshot.cn/v1",
        )
        resp = client.chat.completions.create(
            model="kimi-k2.6",
            temperature=1,
            max_tokens=4000,
            messages=[{
                "role": "user",
                "content": [
                    {"type": "image_url", "image_url": {"url": f"data:image/jpeg;base64,{b64}"}},
                    {"type": "text", "text": AI_RECOGNIZE_PROMPT},
                ],
            }],
        )
        raw = resp.choices[0].message.content.strip()
        # 去掉 markdown ```json ... ``` 包裹
        if raw.startswith("```"):
            raw = "\n".join(raw.split("\n")[1:]).rsplit("```", 1)[0]
        data = json.loads(raw)
        return {"success": True, **data, "error": None}
    except Exception as e:
        return {"success": False, "source_order_no": None, "customer": None,
                "dest_address": None, "items": [], "error": str(e)}
```

# models/_permissions.py
from enum import Enum

class Action(str, Enum):
    CREATE_TASK = "create_task"
    EDIT_TASK = "edit_task"
    ASSIGN = "assign"
    CODING_CLAIM = "coding_claim"
    CODING_DONE = "coding_done"
    CODING_RELEASE = "coding_release"
    LOAD = "load"          # 装货(含出发)
    ARRIVE = "arrive"
    UNLOAD = "unload"
    COUNT = "count"
    DRIVER_RELEASE = "driver_release"
    TASK_CANCEL = "task_cancel"
    RETURN_CREATE = "return_create"

def can(operator, action: Action, task=None) -> bool:
    """集中权限判断。operator 是 Staff 实例(或 dict 含 role/id), task 是 Task 实例。"""
    if operator is None:
        return False
    role = getattr(operator, "role", None) or operator.get("role") if isinstance(operator, dict) else None
    op_id = getattr(operator, "id", None) or operator.get("id") if isinstance(operator, dict) else None
    if role is None:
        return False

    # 任意登录者(司机/调度/文员/搬运/打码/仓管) 可建单/编辑
    if action in (Action.CREATE_TASK, Action.EDIT_TASK):
        return role in ("司机", "调度", "文员", "搬运", "打码", "仓管")

    # 打码三操作
    if action in (Action.CODING_CLAIM, Action.CODING_DONE, Action.CODING_RELEASE):
        return role == "打码"

    # 指派司机
    if action == Action.ASSIGN:
        return role in ("调度", "文员")

    # 点数
    if action == Action.COUNT:
        return role in ("文员", "仓管")

    # 装/到/卸
    if action in (Action.LOAD, Action.ARRIVE, Action.UNLOAD, Action.RETURN_CREATE):
        if role != "司机":
            return False
        if task is None:
            return False
        driver_id = getattr(task, "driver_id", None)
        if isinstance(task, dict):
            driver_id = task.get("driver_id")
        return driver_id == op_id

    # 司机退单
    if action == Action.DRIVER_RELEASE:
        if role != "司机" or task is None:
            return False
        driver_id = getattr(task, "driver_id", None) or (task.get("driver_id") if isinstance(task, dict) else None)
        status = getattr(task, "status", None) or (task.get("status") if isinstance(task, dict) else None)
        return driver_id == op_id and status in ("准备中", "已装货")

    # 任务作废
    if action == Action.TASK_CANCEL:
        if role not in ("调度", "文员") or task is None:
            return False
        status = getattr(task, "status", None) or (task.get("status") if isinstance(task, dict) else None)
        return status == "准备中"

    return False
```

from models._permissions import can, Action
from blueprints._helpers import recognize_order_image, get_upload_dir

bp = Blueprint("task_flow", __name__)

def _current_operator():
    op_id = session.get("operator_id")
    return Staff.get_by_id(op_id) if op_id else None

@bp.route("/api/v1/tasks", methods=["POST"])
def create_task():
    op = _current_operator()
    if not can(op, Action.CREATE_TASK):
        return jsonify(success=False, error="无权限建单"), 403
    data = request.get_json(force=True, silent=True) or {}
    try:
        tid = Task.create(
            task_no=data["task_no"],
            creator_id=op["id"],
            source_order_no=data.get("source_order_no"),
            source_type=data.get("source_type"),
            customer=data.get("customer"),
            dest_address=data.get("dest_address"),
        )
    except ValueError as e:
        return jsonify(success=False, error=str(e)), 400
    for i, item in enumerate(data.get("items", [])):
        TaskItem.create(
            task_id=tid,
            product_name=item["product_name"],
            specification=item.get("specification"),
            quantity=item["quantity"],
            unit=item.get("unit", "y"),
            remark=item.get("remark"),
            sort_order=i,
        )
    TaskEvent.create(task_id=tid, event_type="assign", operator_id=op["id"],
                      from_status=None, to_status="准备中", note="建单")
    return jsonify(success=True, task_id=tid)

@bp.route("/tasks", methods=["GET"])
def list_view():
    tasks = Task.get_all(include_cancelled=False)
    return render_template("tasks.html", tasks=tasks)

@bp.route("/tasks/<int:tid>", methods=["GET"])
def detail_view(tid):
    t = Task.get_by_id(tid)
    if not t: abort(404)
    return render_template("task-detail.html", task=t,
                           items=TaskItem.get_by_task(tid),
                           events=TaskEvent.get_by_task(tid))

ALLOWED_TRANSITIONS = {
    "准备中": ["已装货"],
    "已装货": ["已点数"],
    "已点数": ["已到达"],
    "已到达": ["已卸货"],
    "已卸货": ["已完成"],
}
EVIDENCE_GATES = {
    ("已装货", "已点数"): ("装车照",),
    ("已点数", "已到达"): ("点数标签照", "点数整体照"),
    ("已到达", "已卸货"): ("卸货照",),
    ("已卸货", "已完成"): ("签收单",),
}
GATE_TO_ACTION = {
    "已装货": Action.LOAD,
    "已点数": Action.COUNT,
    "已到达": Action.ARRIVE,
    "已卸货": Action.UNLOAD,
}

@bp.route("/api/v1/tasks/<int:tid>/advance", methods=["POST"])
def advance(tid):
    op = _current_operator()
    t = Task.get_by_id(tid)
    if not t: return jsonify(success=False, error="任务不存在"), 404
    to = (request.get_json(silent=True) or {}).get("to_status")
    if to not in ALLOWED_TRANSITIONS.get(t["status"], []):
        return jsonify(success=False, error=f"非法跳转 {t['status']} → {to}"), 400
    # 闸门:装货前置
    if t["status"] == "准备中" and to == "已装货":
        if not t["driver_id"]:
            return jsonify(success=False, error="未指派司机"), 400
        if t["coding_status"] not in ("无需打码", "打码完成"):
            return jsonify(success=False, error="打码未完成,禁止装货"), 400
        # 司机本人 + 装货=出发 → 写 depart_at
        from datetime import datetime
        Task.update(tid, depart_at=datetime.now().strftime("%Y-%m-%d %H:%M:%S"))
    # 闸门:证据
    stages = EVIDENCE_GATES.get((t["status"], to))
    if stages:
        imgs = TaskImage.get_by_task(tid)
        have = {i["stage"] for i in imgs}
        if not any(s in have for s in stages):
            return jsonify(success=False, error=f"缺证据照(需要 {stages} 之一)"), 400
    # 动作权限
    action = GATE_TO_ACTION.get(t["status"])
    if action and not can(op, action, t):
        return jsonify(success=False, error="无权限"), 403
    # 推进
    from datetime import datetime
    extra = {}
    if to == "已到达":
        extra["arrive_at"] = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    Task.update(tid, status=to, **extra)
    TaskEvent.create(tid, "advance", operator_id=op["id"],
                     from_status=t["status"], to_status=to, note=request.json.get("note"))
    return jsonify(success=True)
```

@bp.route("/api/v1/tasks/<int:tid>/driver-release", methods=["POST"])
def driver_release(tid):
    op = _current_operator()

    TaskEvent.create(tid, "driver_release", operator_id=op["id"],
                     from_status=t["status"], to_status="准备中", note="司机退单")

    op = _current_operator()
    t = Task.get_by_id(tid)
    if not can(op, Action.TASK_CANCEL, t):
